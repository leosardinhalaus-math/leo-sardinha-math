cmake_minimum_required(VERSION 3.24)
project(VintageCam VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
if(MSVC)
 add_compile_options(/utf-8 /EHsc /W4 /permissive- /Zc:__cplusplus)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(CoreTests tests/CoreTests.cpp src/Film.cpp)
target_include_directories(CoreTests PRIVATE src)
add_test(NAME CoreTests COMMAND CoreTests)
add_executable(CameraTests tests/CameraTests.cpp)
target_include_directories(CameraTests PRIVATE src)
add_test(NAME CameraTests COMMAND CameraTests)
if(WIN32)
 if(NOT MSVC)
  message(FATAL_ERROR "Use MSVC 2022 e Windows SDK com C++/WinRT.")
 endif()
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_compile_definitions(UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
 add_compile_options(/utf-8 /EHsc /W4 /permissive- /Zc:__cplusplus)
 set(WIN_LIBS windowsapp runtimeobject d3d11 dxgi d3dcompiler windowscodecs shlwapi shell32 ole32 oleaut32 user32 gdi32 advapi32 mf mfplat mfreadwrite mfuuid)
 add_executable(VintageCam WIN32 src/App.cpp src/Camera.cpp src/CompatCamera.cpp src/Gpu.cpp src/ImageIO.cpp src/Share.cpp app.manifest)
 target_include_directories(VintageCam PRIVATE src)
 target_link_libraries(VintageCam PRIVATE ${WIN_LIBS})
 add_executable(AsyncTests tests/AsyncTests.cpp)
 target_include_directories(AsyncTests PRIVATE src)
 target_link_libraries(AsyncTests PRIVATE ${WIN_LIBS})
 add_test(NAME AsyncTests COMMAND AsyncTests)
 add_executable(GpuTests tests/GpuTests.cpp src/Gpu.cpp src/ImageIO.cpp)
 target_include_directories(GpuTests PRIVATE src)
 target_link_libraries(GpuTests PRIVATE ${WIN_LIBS})
 add_test(NAME GpuTests COMMAND GpuTests)
endif()
